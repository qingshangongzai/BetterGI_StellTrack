```python
def convert_time_to_ms(value: float, unit: str) -> int:
    """将时间值转换为毫秒
    
    Args:
        value: 时间值
```

```python
def handle_errors(logger=None, error_title="错误"):
    """统一的错误处理装饰器"""
    def decorator(func):
        def wrapper(*args, **kwargs):
            try:
                return func(*args, **kwargs)
            except Exception as e:
                # 统一的错误处理逻辑
                if logger:
                    logger.log_error(f"{error_title}: {str(e)}", exc_info=True)
                traceback.print_exc()
        return wrapper
    return decorator
```
